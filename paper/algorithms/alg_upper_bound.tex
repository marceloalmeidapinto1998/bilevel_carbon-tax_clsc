\tcp*[h]{Evaluate the follower exactly at the master's current candidate $\taxindex^{(k)}$}
Solve~\eqref{eq:follower} at $\taxindex=\taxindex^{(k)}$ to optimality, and
among its optimal responses select the one with least emissions
(\Cref{as:opt}), giving $\bz^{(k)}\in\Sol{\taxindex^{(k)}}$,
$c_{k}\defeq\Cost\!\bigl(\bz^{(k)}\bigr)$,
$e_{k}\defeq\Emis\!\bigl(\bz^{(k)}\bigr)$\;
\tcp*[h]{Lexicographic incumbent update: strictly lower emissions, or a
  tie broken toward the smaller tax index -- mirrors the leader's own
  objective~\eqref{eq:leader_obj}}
\If{$e_{k}<\UB$ \textup{\textbf{or}} $\bigl(e_{k}=\UB
    \textup{ \textbf{and} } \taxindex^{(k)}<\taxindex^{\dagger}\bigr)$}{%
  $\UB\leftarrow e_{k}$;\quad
  $\taxindex^{\dagger}\leftarrow\taxindex^{(k)}$;\quad
  $\bz^{\dagger}\leftarrow\bz^{(k)}$\;
}
\tcp{Record the cut
  $\Cost(\bz)+\gridstep\,\taxindex\,\Emis(\bz)\le c_{k}+\gridstep\,\taxindex\,e_{k}$,
  valid for every $\taxindex\in\mathcal{T}^{\tau}$ since
  $\bz^{(k)}\in\setX$ regardless of $\taxindex$ (\Cref{lem:lb_validity})}
$\mathcal{C}\leftarrow\mathcal{C}\cup\bigl\{(\taxindex^{(k)},\,c_{k},\,e_{k})\bigr\}$\;
